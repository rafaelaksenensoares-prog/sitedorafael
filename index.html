using System.Collections.Generic; using UnityEngine; public class GeradorDeMoveis : MonoBehaviour { [System.Serializable] public struct PontoDeSpawn { public Transform localizacao; public List opcoesDeMoveis; // Móveis válidos para este local } [Header("Locais de Spawn dentro da Casa")] public List pontosDaCasa; void Start() { DecorarCasa(); } public void DecorarCasa() { foreach (PontoDeSpawn ponto in pontosDaCasa) { if (ponto.opcoesDeMoveis.Count > 0 && ponto.localizacao != null) { // Escolhe um móvel aleatório da lista para aquele ponto int indiceAleatorio = Random.Range(0, ponto.opcoesDeMoveis.Count); GameObject movelecolhido = ponto.opcoesDeMoveis[indiceAleatorio]; // Instancia o móvel na posição e rotação do ponto Instantiate(movelecolhido, ponto.localizacao.position, ponto.localizacao.rotation, ponto.localizacao); } } } }
