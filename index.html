using System.Collections; using System.Collections.Generic; using UnityEngine; // --- DEFINIÇÃO DAS CLASSES DE PEIXES --- public enum DificuldadePeixe { Facil, Medio, Dificil, Lendario } [System.Serializable] public class Peixe { public string nome; public DificuldadePeixe dificuldade; public float valorVenda; public float forcaResistencia; // O quão difícil é puxar este peixe } // --- SISTEMA PRINCIPAL DO JOGO --- public class SimuladorDePesca : MonoBehaviour { [Header("Configurações do Jogador e Isca")] public Transform pontoDeLancamento; // Ponto na câmera onde a isca sai public GameObject iscaPrefab; public float forcaLancamento = 15f; public float velocidadeRecolhimento = 5f; [Header("Economia e Upgrades")] public float dinheiro = 0f; public int nivelVara = 1; // Facilita puxar peixes difíceis public int nivelBarco = 1; [Header("Banco de Dados de Peixes")] public List tiposDePeixes; // Variáveis de Controle de Estado private GameObject iscaAtual; private bool iscaNaAgua = false; private Peixe peixeFisgado = null; private bool pescando = false; void Update() { // 1. JOGAR A ISCA (Mouse - Botão Esquerdo) if (Input.GetMouseButtonDown(0) && iscaAtual == null) { LancarIsca(); } // 2. PUXAR A ISCA / PEIXE (Barra de Espaço) if (Input.GetKey(KeyCode.Space) && iscaAtual != null) { RecolherIsca(); } // 3. VENDER PEIXE (Exemplo de Loja ao apertar V) if (Input.GetKeyDown(KeyCode.V)) { ComprarUpgradeVara(); } } void LancarIsca() { pescando = true; iscaNaAgua = false; peixeFisgado = null; // Cria a isca e joga ela para frente iscaAtual = Instantiate(iscaPrefab, pontoDeLancamento.position, pontoDeLancamento.rotation); Rigidbody rb = iscaAtual.GetComponent(); if (rb != null) { rb.AddForce(pontoDeLancamento.forward * forcaLancamento, ForceMode.Impulse); } // Inicia a rotina para esperar a isca cair na água e fisgar o peixe StartCoroutine(EsperarPeixeMorder()); } void RecolherIsca() { // Calcula a velocidade que a isca volta pro jogador float velocidadeFinal = velocidadeRecolhimento; // Se tem um peixe, a resistência dele diminui a velocidade, mas o nível da vara ajuda if (peixeFisgado != null) { float dificuldadeAtual = peixeFisgado.forcaResistencia - (nivelVara * 2); if (dificuldadeAtual < 1) dificuldadeAtual = 1; // Garante que nunca fique impossível velocidadeFinal /= dificuldadeAtual; } // Move a isca de volta para o jogador iscaAtual.transform.position = Vector3.MoveTowards( iscaAtual.transform.position, pontoDeLancamento.position, velocidadeFinal * Time.deltaTime ); // Se a isca chegou de volta ao jogador if (Vector3.Distance(iscaAtual.transform.position, pontoDeLancamento.position) < 1.5f) { if (peixeFisgado != null) { Debug.Log($"Você pegou um {peixeFisgado.nome}! Ganhou R$ {peixeFisgado.valorVenda}"); dinheiro += peixeFisgado.valorVenda; } else { Debug.Log("Você recolheu a isca vazia."); } Destroy(iscaAtual); iscaAtual = null; pescando = false; } } IEnumerator EsperarPeixeMorder() { // Espera 2 segundos para simular a isca caindo e estabilizando na água yield return new WaitForSeconds(2f); iscaNaAgua = true; Debug.Log("A isca está na água... aguardando peixe."); // Espera um tempo aleatório entre 3 a 8 segundos para o peixe morder float tempoEspera = Random.Range(3f, 8f); yield return new WaitForSeconds(tempoEspera); // Se o jogador não puxou a isca antes do tempo if (iscaAtual != null && iscaNaAgua) { // Sorteia um peixe da lista baseado no nível do barco (barco melhor = peixes melhores) int indexAleatorio = Random.Range(0, Mathf.Min(tiposDePeixes.Count, nivelBarco * 2)); peixeFisgado = tiposDePeixes[indexAleatorio]; Debug.Log($"FISGOU! Um {peixeFisgado.nome} mordeu a isca! Segure ESPAÇO para puxar!"); // Efeito visual na isca (faz a isca afundar um pouco) Rigidbody rb = iscaAtual.GetComponent(); if(rb != null) rb.AddForce(Vector3.down * 5f, ForceMode.Impulse); } } // --- SISTEMA DE UPGRADES --- void ComprarUpgradeVara() { float custoUpgrade = nivelVara * 100f; // Ex: Nível 1 custa 100, Nível 2 custa 200... if (dinheiro >= custoUpgrade) { dinheiro -= custoUpgrade; nivelVara++; Debug.Log($"Vara aprimorada para o Nível {nivelVara}! Nova força de puxada."); } else { Debug.Log($"Dinheiro insuficiente. Custa R$ {custoUpgrade}. Você tem R$ {dinheiro}."); } } }
